```javascript
// 平台检测
function detectPlatform() {
  const userAgent = navigator.userAgent;
  
  // 微信检测
  const isWeChat = /MicroMessenger/i.test(userAgent);
  
  // 抖音检测
  const isDouyin = /aweme/i.test(userAgent);
  
  // 小红书检测
  const isXinghongshu = /Xinghongshu/i.test(userAgent);
  
  // 桌面检测
  const isDesktop = !/Mobile|Android|iPhone/i.test(userAgent);
  
  return { isWeChat, isDouyin, isXinghongshu, isDesktop };
}

// 性能监控
function monitorPerformance() {
  const performance = window.performance;
  const timing = performance.timing;
  
  const metrics = {
    // 网络相关
    dnsLookup: timing.domainLookupEnd - timing.domainLookupStart,
    tcpConnect: timing.connectEnd - timing.connectStart,
    request: timing.responseStart - timing.requestStart,
    response: timing.responseEnd - timing.responseStart,
    
    // 渲染相关
    domLoad: timing.domComplete - timing.domInteractive,
    windowLoad: timing.loadEventEnd - timing.navigationStart,
    
    // 总体
    totalTime: timing.loadEventEnd - timing.navigationStart
  };
  
  // 发送监控数据
  sendPerformanceData(metrics);
}
```
