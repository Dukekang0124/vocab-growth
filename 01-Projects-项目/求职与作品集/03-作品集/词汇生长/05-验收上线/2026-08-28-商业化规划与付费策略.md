## 一、商业化战略概述

### 1.1 商业化目标

**短期目标（0-3 个月）**：
- 验证产品市场需求（目标：1000+ 免费用户）
- 收集用户反馈（目标：500+ 反馈）
- 付费转化率 > 5%（目标：50+ 付费用户）

**中期目标（3-12 个月）**：
- 稳定付费用户增长（目标：200+ 付费用户）
- 月度收入 > ¥5,000
- 用户留存率 > 30%

**长期目标（12 个月+）**：
- 稳定月度收入 > ¥20,000
- 用户留存率 > 50%
- 品牌知名度提升

---

### 1.2 商业化原则

1. **用户价值优先**：付费功能必须解决真实痛点，带来明显价值
2. **免费版足够好用**：免费版提供完整核心功能，让用户愿意长期使用
3. **渐进式转化**：通过免费体验建立信任，再引导付费转化
4. **数据驱动决策**：用用户行为数据指导产品迭代和付费策略

---

### 1.3 商业化路径

```
免费体验版（v2.0.0）
```

```javascript
// js/usage-limit.js
const USAGE_LIMIT = {
  dailyReview: 10,
  dailySentence: 5,
  dailyNewWord: 3,
  dailyStudyTime: 30 * 60 * 1000 // 30 分钟
};

async function checkUsageLimit(userId) {
  const today = new Date().toISOString().split('T')[0];

  // 检查每日复习次数
  const reviewCount = await getReviewCount(userId, today);
  if (reviewCount >= USAGE_LIMIT.dailyReview) {
    return { allowed: false, reason: '每日复习次数已达上限' };
  }

  // 检查每日造句次数
  const sentenceCount = await getSentenceCount(userId, today);
  if (sentenceCount >= USAGE_LIMIT.dailySentence) {
    return { allowed: false, reason: '每日造句次数已达上限' };
  }

  // 检查每日添加新词次数
  const newWordCount = await getNewWordCount(userId, today);
  if (newWordCount >= USAGE_LIMIT.dailyNewWord) {
    return { allowed: false, reason: '每日添加新词已达上限' };
  }

  return { allowed: true };
}

async function recordUsage(userId, action, today) {
  const key = `${userId}_${action}_${today}`;
  const count = await getUsageCount(key);
  await setUsageCount(key, count + 1);
}
```

##### 2.1.2.2 学习时长限制

**实现方式**：
```javascript
// js/study-time-tracking.js
let studyStartTime = null;
let studyEndTime = null;

async function startStudy(userId) {
  studyStartTime = Date.now();
  await recordStudyStart(userId);
}

async function endStudy(userId) {
  studyEndTime = Date.now();
  const studyDuration = studyEndTime - studyStartTime;

  if (studyDuration > USAGE_LIMIT.dailyStudyTime) {
    return { allowed: false, reason: '今日学习时长已达上限' };
  }

  await recordStudyEnd(userId, studyDuration);
  return { allowed: true };
}

function getRemainingStudyTime() {
  if (!studyStartTime) return USAGE_LIMIT.dailyStudyTime;

  const elapsed = Date.now() - studyStartTime;
  return Math.max(0, USAGE_LIMIT.dailyStudyTime - elapsed);
}
```
