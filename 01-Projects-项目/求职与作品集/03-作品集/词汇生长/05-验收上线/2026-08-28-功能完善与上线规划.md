```javascript
// js/auth.js
const API_BASE = 'https://vocab-growth-api.your-domain.com';

// 登录
async function login(email, password) {
  try {
    const response = await fetch(`${API_BASE}/api/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });

    if (!response.ok) {
      throw new Error('账号或密码错误');
    }

    const data = await response.json();
    localStorage.setItem('userId', data.userId);
    localStorage.setItem('token', data.token);
    localStorage.setItem('username', data.email);
    return true;
  } catch (error) {
    showError(error.message);
    return false;
  }
}

// 注册
async function register(email, password) {
  try {
    const response = await fetch(`${API_BASE}/api/register`, {

      headers: { 'Content-Type': 'application/json' },

    });

    if (!response.ok) {
      throw new Error('注册失败，邮箱可能已被使用');
    }

    const data = await response.json();

}

  localStorage.removeItem('userId');
```

**步骤 3：修改主页支持登录状态**
```javascript
// js/store.js
const store = {
  state: {
    isLogin: false,
    userId: null,
    username: null,
    syncStatus: 'synced'
  },

  async init() {
    // 检查登录状态
    const userId = localStorage.getItem('userId');
    const token = localStorage.getItem('token');

    if (userId && token) {
      this.state.isLogin = true;
      this.state.userId = userId;
      this.state.username = localStorage.getItem('username');
      await this.syncData();
    }
  },

  async syncData() {
    this.state.syncStatus = 'syncing';
    try {
      const response = await fetch(`${API_BASE}/api/sync`, {
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        }
      });

      if (response.ok) {
        const data = await response.json();
        this.state.syncStatus = 'synced';
        return data;
      }
    } catch (error) {
      this.state.syncStatus = 'error';
      console.error('同步失败:', error);
    }
  }
};
```

**步骤 2：实现登录接口**
```javascript
async function handleLogin(request, env) {
  const body = await request.json();
  const { email, password } = body;

  // 验证邮箱格式
  if (!isValidEmail(email)) {
    return new Response(JSON.stringify({ error: '邮箱格式错误' }), {
      status: 400,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  // 从 KV 读取用户
  const userId = hashEmail(email);
  const user = await env.VOCAB_GROWTH_USERS.get(userId);

  if (!user) {
    return new Response(JSON.stringify({ error: '账号不存在' }), {
      status: 404,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  // 验证密码
  const passwordHash = await sha256(password);
  if (user.passwordHash !== passwordHash) {
    return new Response(JSON.stringify({ error: '密码错误' }), {
      status: 401,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  // 更新最后登录时间
  const updatedUser = {
    ...user,
    lastLoginAt: new Date().toISOString()
  };
  await env.VOCAB_GROWTH_USERS.put(userId, JSON.stringify(updatedUser));

  // 返回 token
  const token = generateToken(userId);
  return new Response(JSON.stringify({
    userId,
    token,
    email
  }), {
    status: 200,
    headers: { 'Content-Type': 'application/json' }
  });
}
```

**步骤 3：实现注册接口**
```javascript
async function handleRegister(request, env) {
  const body = await request.json();
  const { email, password } = body;

  // 验证邮箱格式
  if (!isValidEmail(email)) {
    return new Response(JSON.stringify({ error: '邮箱格式错误' }), {
      status: 400,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  // 验证密码强度
  if (password.length < 6) {
    return new Response(JSON.stringify({ error: '密码长度至少 6 位' }), {
      status: 400,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  // 检查邮箱是否已注册
  const userId = hashEmail(email);
  const existingUser = await env.VOCAB_GROWTH_USERS.get(userId);

  if (existingUser) {
    return new Response(JSON.stringify({ error: '邮箱已被注册' }), {
      status: 400,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  // 创建新用户
  const newUser = {
    userId,
    email,
    passwordHash: await sha256(password),
    createdAt: new Date().toISOString(),
    lastLoginAt: new Date().toISOString(),
    dataVersion: 1
  };

  await env.VOCAB_GROWTH_USERS.put(userId, JSON.stringify(newUser));

  // 初始化用户数据表
  await env.VOCAB_GROWTH_DATA.put(`${userId}_data`, JSON.stringify({
    userId,
    version: 1,
    updatedAt: new Date().toISOString(),
    data: getSeedData() // 从种子数据初始化
  }));

  // 返回 token
  const token = generateToken(userId);
  return new Response(JSON.stringify({
    userId,
    token,
    email
  }), {
    status: 200,
    headers: { 'Content-Type': 'application/json' }
  });
}
```

**步骤 4：实现数据同步接口**
```javascript
async function handleSync(request, env) {
  const body = await request.json();
  const { userId, data } = body;

  // 验证 token
  const authHeader = request.headers.get('Authorization');
  const token = authHeader?.replace('Bearer ', '');

  if (!token || !verifyToken(token, userId)) {
    return new Response(JSON.stringify({ error: '未授权' }), {
      status: 401,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  // 更新用户数据
  await env.VOCAB_GROWTH_DATA.put(`${userId}_data`, JSON.stringify({
    userId,
    version: data.version + 1,
    updatedAt: new Date().toISOString(),
    data
  }));

  // 记录同步日志
  const log = {
    timestamp: new Date().toISOString(),
    action: 'update',
    dataVersion: data.version
  };

  const syncLog = await env.VOCAB_GROWTH_SYNC_LOG.get(`${userId}_sync_log`);
  const logs = syncLog ? JSON.parse(syncLog) : [];
  logs.push(log);
  await env.VOCAB_GROWTH_SYNC_LOG.put(`${userId}_sync_log`, JSON.stringify(logs));

  return new Response(JSON.stringify({ success: true }), {
    status: 200,
    headers: { 'Content-Type': 'application/json' }
  });
}
```

```javascript
// sw.js
self.addEventListener('install', event => {
  event.waitUntil(
    caches.open('vocab-growth-v1').then(cache => {
      return cache.addAll([
        './',
        './index.html',
        './css/style.css',
        './js/app.js',
        './js/data.js',
        './js/srs.js',
        './js/store.js',
        './js/auth.js',
        './assets/audio/*.mp3'
      ]);
    })
  );
});

self.addEventListener('fetch', event => {
  event.respondWith(
    caches.match(event.request).then(response => {
      return response || fetch(event.request);
    })
  );
});
```

3. **注册 service worker**
```javascript
if ('serviceWorker' in navigator) {
  navigator.serviceWorker.register('./sw.js')

}
```
