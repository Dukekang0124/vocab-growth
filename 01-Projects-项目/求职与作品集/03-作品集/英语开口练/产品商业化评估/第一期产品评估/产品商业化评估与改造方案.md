### 2.3 必须正视的一个策略冲突

29.9 买断制和"加微信引流私域转化"**不可能同时最大化**：付了钱的用户没有理由再加你微信。既然目标是真赚钱，把关系理顺为：

```
¥29.9 产品（本APP）   =   低门槛信任敲门砖 + 自动收租工具
     ↓ 付费用户才被导流
微信社群/训练营        =   利润主力（未来 ¥99~199 训练营），29.9 是获客成本
```

所以"关于页"的作用要从"引流卖课"改为"**信任背书 + 售后入口**"，付费墙里加微信的动作从"为了卖你课"变为"给你发码、售后、进学员群"。这个叙事变了，文案全都要跟着变（见 P1-2）。

### P0-1 修好收银台（最高优先级，2小时）

#### A. 加"复制微信号"按钮

这是全案性价比最高的一个改动。在付费弹窗第 2 步下方（`index.html` 约 664 行，`pay-steps` 内第二个 `.step` 之后）加：

```html
<button class="btn btn-primary btn-sm" style="width:auto;margin:10px auto 0;display:block;"
        onclick="copyWechat()">复制微信号 SK-Duke</button>
```

JS（放在 `tryUnlock` 函数附近）：

```js
function copyWechat(){
  var wxid='SK-Duke';
  function done(){
    showToast('已复制 '+wxid+'，请打开微信-添加朋友-粘贴');
    if(window._hmt){_hmt.push(['_trackEvent','funnel','copy_wechat']);}
  }
  function fallback(){
    var i=document.createElement('input');
    i.value=wxid;document.body.appendChild(i);i.select();
    try{document.execCommand('copy');done();}
    catch(e){showToast('复制失败，请手动输入：'+wxid);}
    document.body.removeChild(i);
  }
  if(navigator.clipboard&&navigator.clipboard.writeText){
    navigator.clipboard.writeText(wxid).then(done).catch(fallback);
  }else{fallback();}
}
```

> 为什么必须保留 `execCommand` 兜底：抖音/微信 webview 里 `navigator.clipboard` 经常被禁，老 API 反而最稳。
>
> 另外 `SK-Duke` 大小写混合+连字符，中年用户手动输入极易打错。**强烈建议顺手注册一个纯小写字母或纯数字的备用微信号用于收款**（如 suabujuan88 这类），至少在文案里补一句"微信号不区分大小写"。

### P0-2 解锁码从明文改为"一单一码"（半天，含生成脚本）

现在 `UNLOCK_CODE='kaikou99'`（约1696行）有三个问题：①人人可知（会被截图传播）；②你对账困难（不知道谁买了）；③一个码全网通用无法作废。

**改成哈希一单一码**：本地预生成一批码，前端只存码的 SHA-256 哈希。每个买家拿到唯一码，可追溯、可单独作废、源码里搜不到任何有效码。

第一步，本地运行生成脚本（新建 `gen-codes.js`，Node 运行）：

```js
const crypto=require('crypto'),fs=require('fs');
const N=200; // 预生成200个码，用完再生成第二批
const out=[];
for(let i=0;i<N;i++){
  const raw=crypto.randomBytes(6).toString('hex').toUpperCase();
  const code=`KK${raw.slice(0,4)}-${raw.slice(4,8)}-${raw.slice(8,12)}`;
  const hash=crypto.createHash('sha256').update(code).digest('hex');
  out.push({code,hash});
}
fs.writeFileSync('codes.json',JSON.stringify(out,null,2)); // 你自己留着：卖一份给一个
fs.writeFileSync('hashes.txt',out.map(r=>"'"+r.hash+"',").join('\n')); // 贴进前端
console.log('OK,共'+N+'个码');
```

第二步，替换前端 `tryUnlock`（约1741行）并把 `hashes.txt` 内容填入第一行数组：

```js
var UNLOCK_HASHES=[ /* 这里贴 hashes.txt 全部内容 */
];
async function tryUnlock(){
  var codeInput=document.getElementById('unlockCodeInput').value.trim().toUpperCase();
  var msg=document.getElementById('unlockMsg');
  if(codeInput==='KAIOUKOU99'||codeInput==='KAIKOU99'){ // 调试期保留万能码，上线一个月后删除
    accept(codeInput,msg);return;
  }
  if(!window.crypto||!crypto.subtle){msg.textContent='请在正常网络环境下解锁';msg.className='unlock-msg error';return;}
  try{
    var buf=await crypto.subtle.digest('SHA-256',new TextEncoder().encode(codeInput));
    var hex=[].slice.call(new Uint8Array(buf)).map(function(b){return b.toString(16).padStart(2,'0')}).join('');
    if(UNLOCK_HASHES.indexOf(hex)>-1){accept(codeInput,msg);}
    else{msg.textContent='解锁码不正确，请确认复制完整，或加微信 SK-Duke 获取';msg.className='unlock-msg error';}
  }catch(e){msg.textContent='解锁异常，请加微信 SK-Duke';msg.className='unlock-msg error';}
}
function accept(code,msg){
  state.unlocked=true;state.unlockCode=code;saveState(); // 记录用的哪个码=你的台账
  msg.textContent='解锁成功！全部内容已开放';msg.className='unlock-msg success';
  setTimeout(function(){closePaywall();showToast('全部功能已解锁！');renderWord();renderDialogScenes();renderRepeat();},1200);
}
```

> 注意两点：① `crypto.subtle` 只在 HTTPS 下可用——按 P0-6 上线后没问题；② `loadState()` 读入旧档案时补一句默认值 `if(state.unlockCode===undefined)state.unlockCode=null;`
>
> **诚实交代防护上限**：这套改动拦不住懂控制台的人（在 Console 执行 `localStorage.setItem('english_speaking_app_v1',JSON.stringify(Object.assign(JSON.parse(localStorage.getItem('english_speaking_app_v1')),{unlocked:true})))` 就绕过了），也拦不住"直接把整份 HTML 发给朋友"。但你的客群是 31-45 岁、从抖音来的零基础学员，会干这两件事的比例极低。**做个算术**：假设流量起来后有 10% 的意向用户技术上可以白嫖且其中一半真的白嫖，损失约等于收入的 5%，而彻底防住需要服务端内容下发（开发 3~5 天 + 服务器长期成本）。正确姿势是：现在不做服务端，把时间投给内容和投流，等月流水过五千再考虑。**把这段写进你的商业计划书，面试/复盘都用得上——这叫"知道墙在哪还敢不砌墙"的判断力。**

### P0-6 腾讯云部署与域名备案（今天启动，周期1~3周）

1. **买域名**：简短全拼或数字，如 `300kaikou.com` / `kklian.cn`（≤8字符，方便视频中口播和评论区手打）。注意要在**腾讯云或转入腾讯云**做 DNS，备案才顺。
2. **提交 ICP 备案**：腾讯云备案控制台，个人主体+博客/工具类目，通常 1~2 周。**备案期间可先用香港轻量服务器临时上线试跑**（无需备案，速度略慢），备案下来再迁大陆节点。
3. **部署形态二选一**：
   - 图省事：COS 静态网站托管 + CDN + 免费 SSL 证书（TrustAsia，3个月自动续），月成本 < ¥5；
   - 要灵活：轻量应用服务器（¥50-60/月档足够）+ Nginx + certbot。纯静态单文件选前者即可。
4. **HTTPS 是硬前提**：麦克风权限（getUserMedia）只在 secure context 下可用， deployment 后第一时间测录音。
5. 部署完成后删掉 aiforce 预览链接对外的露出，正式二维码/链接全部换新域名（自带去掉"豆包AI生成"角标的好处）。

### P1-4 TTS 体验固化（合并两个补丁）

let enVoice=null;
function pickVoice(){
